> the user just spoke in. Keep answers SHORT and conversational (1-3 sentences) since your
> reply will be spoken out loud. Use the provided tools whenever the user asks about the
> time or weather — never guess live data."""
>
> recognizer = sr.Recognizer()
>
>
> def transcribe_audio(audio_path: str) -> str:
>     """Free speech-to-text using Google's Web Speech API (no key needed)."""
>     with sr.AudioFile(audio_path) as source:
>         audio = recognizer.record(source)
>     try:
>         # try English first, then Hindi, then Urdu
>         for lang in ["en-US", "hi-IN", "ur-PK"]:
>             try:
>                 return recognizer.recognize_google(audio, language=lang)
>             except sr.UnknownValueError:
>                 continue
>         return ""
>     except sr.RequestError as e:
>         return f"[STT error: {e}]"
>
>
> def ask_gemini_with_audio(audio_path: str) -> str:
>     """Transcribes the recorded command, then asks Gemini for a short spoken reply."""
>     user_text = transcribe_audio(audio_path)
>     if not user_text:
>         return "Sorry, I didn't catch that."
>
>     response = client.models.generate_content(
>         model="gemini-3.1-flash-lite",
>         contents=user_text,
>         config=types.GenerateContentConfig(
>             system_instruction=SYSTEM_PROMPT,
>             tools=[get_current_time, get_weather],
>         ),
>     )
>     return response.text

> def record_command(out_path: str = "command.wav", max_seconds: int = 8) -> str:
>     """Records audio after the wake word until ~0.5s of silence is detected."""
>     SAMPLE_RATE = 16000
>     FRAME_MS = 30
>     FRAME_SIZE = int(SAMPLE_RATE * FRAME_MS / 1000)
>     vad = webrtcvad.Vad(2)
>
>     frames = []
>     silence_run = 0
>     triggered = False
>     max_frames = int(max_seconds * 1000 / FRAME_MS)
>
>     with sd.InputStream(samplerate=SAMPLE_RATE, channels=1, dtype="int16",
>                          blocksize=FRAME_SIZE, device=MIC_DEVICE_INDEX) as stream:
>         print("Listening for your command...")
>         for _ in range(max_frames):
>             audio, _ = stream.read(FRAME_SIZE)
>             pcm = audio.tobytes()
>             is_speech = vad.is_speech(pcm, SAMPLE_RATE)
>             frames.append(pcm)
>             if is_speech:
>                 triggered = True
>                 silence_run = 0
>             elif triggered:
>                 silence_run += 1
>                 if silence_run > int(500 / FRAME_MS):  # ~0.5s of silence
>                     break
>
>     with wave.open(out_path, "wb") as wf:
>         wf.setnchannels(1)
>         wf.setsampwidth(2)
>         wf.setframerate(SAMPLE_RATE)
>         wf.writeframes(b"".join(frames))
>
>     return out_path
